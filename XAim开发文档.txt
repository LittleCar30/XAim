第一部分：产品与战略
1. 项目概述
1.1 一句话定义
XAim 是一个运行在桌面上的编程 Agent，用户用自然语言描述任务，XAim 自动完成代码定位、修改、验证、回滚，并把过程透明展示给用户。

1.2 名字含义
X = eXplore / eXecute / 未知

Aim = 目标、瞄准

口号：

Aim at intent, eXecute code.
瞄准意图，执行代码。

1.3 产品定位
不是 IDE，不是终端工具，不是聊天机器人。

而是：

桌面上的"编程任务执行器"，带验证、带回滚、带过程透明。

1.4 核心价值
准确：验证驱动，没有验证不算完成

快速：本地索引 + 缓存 + 并行 + 模型路由

透明：上下文、工具调用、Diff 全可见

可控：Git 回滚点，一键恢复

本地优先：代码可不出本机，支持本地模型

2. 目标与原则
2.1 三大目标
优先级	目标	说明
P0	准确	结果必须通过验证
P1	快速	任务完成时间尽可能短
P2	简洁	后置目标，前期允许复杂
2.2 设计原则
没有验证的修改不算完成

每个修改必须可追溯到文件和行号

高风险修改必须人工确认

失败自动回滚，不污染工作区

多候选并行验证，优于单次强模型

本地能力优先，模型能力补充

过程透明，用户可审计

能缓存就缓存，能并行就并行

每次任务一个 Git 回滚点

不确定时询问用户，不猜测

2.3 非目标
第一版明确不做：

完整 IDE

终端模拟器

多 Agent 协作

自动部署

插件市场

移动端

云端协作

代码补全（这是 IDE 的事）

3.3 用户故事
text
作为开发者，
我希望输入一句中文任务，
XAim 自动定位代码、修改、跑测试，
并在失败时自动修复，
最后让我确认 Diff，
这样我不用手动搜索和反复调试。

5. 看到步骤时间线
6. 看到候选 Diff
7. 看到验证结果
8. 点击应用
9. 任务完成
10. 不满意可一键回滚

6.2 进程模型
text
主进程：Tauri App
  ├── UI 渲染进程（WebView）
  ├── Rust 核心（主进程内，async runtime）
  ├── 索引子进程（可选，独立线程）
  ├── 命令执行子进程（每次任务独立）
  └── 模型请求（异步 HTTP / 本地）
6.3 数据流
text
用户输入
  -> UI
  -> Tauri IPC (invoke)
  -> Rust 任务编排器
  -> 检索 + 上下文构建
  -> 模型调用（流式）
  -> 工具调用（读/写/执行）
  -> 验证
  -> 失败修复循环
  -> Diff 事件流回 UI (emit)
  -> 用户确认
  -> 应用 + Git commit
6.4 线程模型
text
Tokio Runtime（多线程）
  ├── 主任务线程
  ├── 索引线程（独立）
  ├── 文件监听线程（独立）
  ├── 命令执行线程池
  └── 模型请求线程池
6.5 事件总线
text
后端 emit "xaim://event"
  -> 前端 listen
  -> 更新 Zustand store
  -> 组件重渲染

7.2 备选方案
如果团队只会 TypeScript：

Electron + TypeScript

Rust sidecar 做索引和执行

代价：包大（100MB+）、内存高（300MB+）、启动慢（3-5秒）

建议：

如果目标是"快速"，优先 Tauri + Rust。
如果目标是"快速开发"，选 Electron，但性能目标要下调。

9. 数据模型
9.1 SQLite 表
sql
-- 项目
CREATE TABLE projects (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  path TEXT UNIQUE NOT NULL,
  name TEXT NOT NULL,
  created_at INTEGER NOT NULL,
  last_opened_at INTEGER NOT NULL
);

-- 文件
CREATE TABLE files (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  project_id INTEGER NOT NULL,
  path TEXT NOT NULL,
  language TEXT,
  size INTEGER,
  mtime INTEGER,
  hash TEXT,
  summary TEXT,
  indexed_at INTEGER,
  UNIQUE(project_id, path),
  FOREIGN KEY (project_id) REFERENCES projects(id) ON DELETE CASCADE
);
CREATE INDEX idx_files_project ON files(project_id);
CREATE INDEX idx_files_hash ON files(hash);

-- 符号
CREATE TABLE symbols (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  file_id INTEGER NOT NULL,
  name TEXT NOT NULL,
  kind TEXT NOT NULL,
  start_line INTEGER NOT NULL,
  end_line INTEGER NOT NULL,
  FOREIGN KEY (file_id) REFERENCES files(id) ON DELETE CASCADE
);
CREATE INDEX idx_symbols_name ON symbols(name);
CREATE INDEX idx_symbols_file ON symbols(file_id);

-- 任务
CREATE TABLE tasks (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  project_id INTEGER NOT NULL,
  prompt TEXT NOT NULL,
  status TEXT NOT NULL,
  intent TEXT,
  plan TEXT,
  created_at INTEGER NOT NULL,
  finished_at INTEGER,
  duration_ms INTEGER,
  token_input INTEGER,
  token_output INTEGER,
  cost REAL,
  FOREIGN KEY (project_id) REFERENCES projects(id) ON DELETE CASCADE
);
CREATE INDEX idx_tasks_project ON tasks(project_id);
CREATE INDEX idx_tasks_status ON tasks(status);

-- 任务步骤
CREATE TABLE task_steps (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  task_id INTEGER NOT NULL,
  step_type TEXT NOT NULL,
  content TEXT,
  created_at INTEGER NOT NULL,
  FOREIGN KEY (task_id) REFERENCES tasks(id) ON DELETE CASCADE
);
CREATE INDEX idx_steps_task ON task_steps(task_id);

-- 补丁
CREATE TABLE patches (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  task_id INTEGER NOT NULL,
  file_path TEXT NOT NULL,
  diff TEXT NOT NULL,
  applied INTEGER DEFAULT 0,
  score REAL,
  created_at INTEGER NOT NULL,
  FOREIGN KEY (task_id) REFERENCES tasks(id) ON DELETE CASCADE
);
CREATE INDEX idx_patches_task ON patches(task_id);

-- 缓存
CREATE TABLE cache (
  key TEXT PRIMARY KEY,
  value TEXT NOT NULL,
  expire_at INTEGER
);
CREATE INDEX idx_cache_expire ON cache(expire_at);

-- 版本
CREATE TABLE schema_version (
  version INTEGER PRIMARY KEY,
  applied_at INTEGER NOT NULL
);

    pub async fn run_task(&self, project_id: i64, prompt: String) -> Result<i64> {
        // 1. 创建任务记录
        let task_id = self.store.insert_task(project_id, &prompt)?;
        self.emit_step(task_id, "解析意图", "running")?;

        // 2. 解析意图
        let intent = self.parse_intent(&prompt).await?;
        self.store.update_task_intent(task_id, &intent)?;
        self.emit_step(task_id, "解析意图", "done")?;

        // 3. 检查索引
        self.emit_step(task_id, "检查索引", "running")?;
        self.ensure_indexed(project_id).await?;
        self.emit_step(task_id, "检查索引", "done")?;

        // 4. 检索相关文件
        self.emit_step(task_id, "检索相关文件", "running")?;
        let context = self.build_context(&prompt, project_id).await?;
        self.emit_step(task_id, "检索相关文件", "done")?;

        // 5. 生成计划
        self.emit_step(task_id, "生成计划", "running")?;
        let plan = self.generate_plan(&prompt, &context).await?;
        self.store.update_task_plan(task_id, &plan)?;
        self.emit_step(task_id, "生成计划", "done")?;

        // 6. 生成候选 Patch
        self.emit_step(task_id, "生成候选", "running")?;
        let candidates = self.generate_candidates(&prompt, &plan, &context).await?;
        self.emit_step(task_id, "生成候选", "done")?;

        // 7. 应用 + 验证 + 评分
        let best = self.evaluate_candidates(task_id, &candidates).await?;

        // 8. 展示 Diff，等待确认
        self.emit_step(task_id, "等待确认", "waiting")?;

        Ok(task_id)
    }
}

// src-tauri/src/search/grep.rs
pub fn search(root: &Path, query: &str, opts: GrepOptions) -> Result<Vec<GrepMatch>> {
    let mut matches = vec![];
    let re = RegexMatcher::new_line_matcher(query)?;

    let walker = WalkBuilder::new(root)
        .git_ignore(true)
        .build();

    for entry in walker {
        let entry = entry?;
        if !entry.file_type().map_or(false, |t| t.is_file()) {
            continue;
        }
        let path = entry.path();
        let content = std::fs::read_to_string(path)?;
        for (line_no, line) in content.lines().enumerate() {
            if re.find(line.as_bytes()).is_some() {
                matches.push(GrepMatch {
                    path: path.to_string_lossy().into(),
                    line: (line_no + 1) as u32,
                    text: line.to_string(),
                });
                if matches.len() >= opts.max_results {
                    return Ok(matches);
                }
            }
        }
    }

    Ok(matches)
}

// src-tauri/src/search/rank.rs
pub fn rank_results(
    task: &str,
    grep: Vec<GrepMatch>,
    symbols: Vec<Symbol>,
) -> Vec<RankedFile> {
    let mut scores: HashMap<String, f64> = HashMap::new();

    // grep 匹配得分
    for m in grep {
        *scores.entry(m.path.clone()).or_default() += 1.0;
    }

    // 符号匹配得分
    let task_lower = task.to_lowercase();
    for s in symbols {
        if task_lower.contains(&s.name.to_lowercase()) {
            *scores.entry(s.file.clone()).or_default() += 3.0;
        }
    }

    // 排序
    let mut ranked: Vec<_> = scores.into_iter().collect();
    ranked.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap());
    ranked.into_iter().map(|(path, score)| RankedFile { path, score }).collect()
}

// src-tauri/src/search/context.rs
pub fn build_context(
    task: &str,
    ranked: Vec<RankedFile>,
    budget_tokens: usize,
) -> Result<Context> {
    let mut ctx = Context { files: vec![], total_tokens: 0 };

    for r in ranked {
        let content = std::fs::read_to_string(&r.path)?;
        let tokens = estimate_tokens(&content);
        if ctx.total_tokens + tokens > budget_tokens {
            // 截断到函数级
            let truncated = truncate_to_functions(&content, budget_tokens - ctx.total_tokens);
            ctx.files.push(ContextFile {
                path: r.path.clone(),
                content: truncated,
                reason: format!("score: {:.2}", r.score),
            });
            break;
        }
        ctx.total_tokens += tokens;
        ctx.files.push(ContextFile {
            path: r.path,
            content,
            reason: format!("score: {:.2}", r.score),
        });
    }

    Ok(ctx)
}

fn estimate_tokens(text: &str) -> usize {
    // 粗略估算：英文 4 字符 1 token，中文 1.5 字符 1 token
    text.chars().count() / 3
}

// src-tauri/src/tools/patch.rs
pub fn parse_diff(text: &str) -> Result<Vec<PatchCandidate>> {
    let mut candidates = vec![];
    let mut current: Option<PatchCandidate> = None;

    for line in text.lines() {
        if line.starts_with("--- ") {
            if let Some(c) = current.take() {
                candidates.push(c);
            }
            current = Some(PatchCandidate {
                file_path: line[4..].trim_start_matches("a/").to_string(),
                diff: String::new(),
                score: None,
            });
        }
        if let Some(c) = current.as_mut() {
            c.diff.push_str(line);
            c.diff.push('\n');
        }
    }

    if let Some(c) = current {
        candidates.push(c);
    }

    Ok(candidates)
}

pub fn apply_diff(diff: &str, root: &Path) -> Result<()> {
    let patch = diffy::Patch::from_str(diff)?;
    let path = extract_path_from_diff(diff)?;
    let full_path = root.join(path);
    let original = std::fs::read_to_string(&full_path)?;
    let patched = diffy::apply(&original, &patch)?;
    std::fs::write(&full_path, patched)?;
    Ok(())
}

    if root.join("Cargo.toml").exists() {
        config.type_check = Some("cargo check".into());
        config.lint = Some("cargo clippy".into());
        config.test = Some("cargo test".into());
        config.build = Some("cargo build".into());
    }

    if root.join("go.mod").exists() {
        config.type_check = Some("go vet ./...".into());
        config.test = Some("go test ./...".into());
        config.build = Some("go build ./...".into());
    }

    config
}

fn detect_package_manager(root: &Path) -> &'static str {
    if root.join("pnpm-lock.yaml").exists() { "pnpm" }
    else if root.join("yarn.lock").exists() { "yarn" }
    else { "npm" }
}

pub async fn run_verify(
    kind: VerifyKind,
    cmd: &str,
    root: &Path,
) -> Result<VerifyResult> {
    let start = std::time::Instant::now();
    let output = tokio::time::timeout(
        Duration::from_secs(300),
        tokio::process::Command::new("sh")
            .arg("-c")
            .arg(cmd)
            .current_dir(root)
            .output()
    ).await??;

    let passed = output.status.success();
    let stdout = String::from_utf8_lossy(&output.stdout);
    let stderr = String::from_utf8_lossy(&output.stderr);

    Ok(VerifyResult {
        kind: kind.to_string(),
        passed,
        output: format!("{}\n{}", stdout, stderr),
        duration_ms: start.elapsed().as_millis() as i64,
    })
}

pub fn parse_tsc(output: &str) -> Vec<DiagError> {
    let re = Regex::new(r"^(.+)\((\d+),(\d+)\): (error|warning) TS\d+: (.+)$").unwrap();
    output.lines().filter_map(|line| {
        re.captures(line).map(|c| DiagError {
            file: c[1].to_string(),
            line: c[2].parse().unwrap_or(0),
            column: c[3].parse().unwrap_or(0),
            severity: c[4].to_string(),
            message: c[5].to_string(),
        })
    }).collect()
}

pub fn parse_cargo(output: &str) -> Vec<DiagError> {
    let re = Regex::new(r"^error(\[E\d+\])?: (.+)$").unwrap();
    output.lines().filter_map(|line| {
        re.captures(line).map(|c| DiagError {
            file: String::new(),
            line: 0,
            column: 0,
            severity: "error".into(),
            message: c[2].to_string(),
        })
    }).collect()
}

pub async fn repair_loop(
    task_id: i64,
    ctx: &ToolContext,
    model: &dyn ModelProvider,
    max_rounds: u32,

    let mut round = 0;

    loop {
        round += 1;
        if round > max_rounds {
            return Ok(RepairResult::Exceeded);
        }

        // 1. 收集错误
        let verify = run_all_verify(ctx).await?;
        if verify.iter().all(|v| v.passed) {
            return Ok(RepairResult::Success);
        }

        // 2. 构建修复 prompt
        let errors = verify.iter()
            .filter(|v| !v.passed)
            .map(|v| v.output.clone())
            .collect::<Vec<_>>()
            .join("\n");

        let prompt = prompts::repair::build(&errors, &ctx.current_code()?);

            model: "default".into(),

                Message { role: "user".into(), content: prompt },

            temperature: 0.1,
            max_tokens: 4096,
        }).await?;

        // 4. 应用新 patch
        let patches = parse_diff(&response.content)?;
        for p in &patches {
            apply_diff(&p.diff, &ctx.project_root)?;
        }

        // 5. 发事件
        // 6. 继续循环
    }

pub fn create_rollback_point(root: &Path, task_id: i64) -> Result<String> {
    let repo = git2::Repository::open(root)?;
    let sig = git2::Signature::now("XAim", "xaim@local")?;
    let mut index = repo.index()?;
    let tree_id = index.write_tree()?;
    let tree = repo.find_tree(tree_id)?;

    let parent = repo.head().ok().and_then(|h| h.peel_to_commit().ok());
    let parents: Vec<&git2::Commit> = parent.iter().collect();

    let commit_id = repo.commit(
        Some("HEAD"),
        &sig,
        &sig,
        &format!("[XAim] rollback point for task {}", task_id),
        &tree,
        &parents,
    )?;

    Ok(commit_id.to_string())
}

pub fn rollback(root: &Path, commit_id: &str) -> Result<()> {
    let repo = git2::Repository::open(root)?;
    let obj = repo.revparse_single(commit_id)?;
    let commit = obj.peel_to_commit()?;
    repo.reset(&commit.into_object(), git2::ResetType::Hard, None)?;
    Ok(())
}

pub fn diff(root: &Path) -> Result<String> {
    let repo = git2::Repository::open(root)?;
    let diff = repo.diff_index_to_workdir(None, None)?;
    let mut out = String::new();
    diff.print(git2::DiffFormat::Patch, |_, _, line| {
        out.push_str(std::str::from_utf8(line.content()).unwrap_or(""));
        true
    })?;
    Ok(out)
}

// src-tauri/src/store/cache.rs
pub fn get_cache(conn: &Connection, key: &str) -> Result<Option<String>> {
    let now = chrono::Utc::now().timestamp();
    let mut stmt = conn.prepare(
        "SELECT value FROM cache WHERE key = ?1 AND (expire_at IS NULL OR expire_at > ?2)"
    )?;
    let mut rows = stmt.query(params![key, now])?;
    if let Some(row) = rows.next()? {
        Ok(Some(row.get(0)?))
    } else {
        Ok(None)
    }
}

pub fn set_cache(conn: &Connection, key: &str, value: &str, ttl: Option<i64>) -> Result<()> {
    let expire = ttl.map(|t| chrono::Utc::now().timestamp() + t);
    conn.execute(
        "INSERT OR REPLACE INTO cache (key, value, expire_at) VALUES (?1, ?2, ?3)",
        params![key, value, expire],
    )?;
    Ok(())
}

// tests/eval/runner.rs
pub async fn run_eval(tasks_dir: &Path) -> EvalReport {
    let mut report = EvalReport::default();

    for entry in std::fs::read_dir(tasks_dir).unwrap() {
        let task_dir = entry.unwrap().path();
        let task = std::fs::read_to_string(task_dir.join("task.md")).unwrap();

        // 复制 repo 到临时目录
        let tmp = TempDir::new().unwrap();
        copy_dir(&task_dir.join("repo"), tmp.path()).unwrap();

        // 跑任务
        let start = std::time::Instant::now();
        let result = run_task_on(tmp.path(), &task).await;
        let duration = start.elapsed();

        // 验证
        let verify = run_verify_sh(&task_dir.join("verify.sh"), tmp.path()).await;

        report.total += 1;
        if verify.passed {
            report.passed += 1;
        } else {
            report.failed += 1;
        }
        report.details.push(TaskResult { task, duration, verify });
    }

    report
}
